let cameraStream = null; let startingCamera = false; async function startCamera() { if (startingCamera) return; const video = document.getElementById("camera"); const message = document.getElementById("cameraMessage"); const status = document.getElementById("status"); const startButton = document.getElementById("startButton"); const stopButton = document.getElementById("stopButton"); startingCamera = true; startButton.disabled = true; status.textContent = "Status: Connecting camera..."; try { // Stop any old camera stream first if (cameraStream) { cameraStream.getTracks().forEach(track => track.stop()); cameraStream = null; } if (!navigator.mediaDevices?.getUserMedia) { throw new Error( "Camera unavailable. Please use HTTPS and allow camera permission." ); } // Request desktop webcam or front camera cameraStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: { ideal: "user" }, width: { ideal: 1280 }, height: { ideal: 720 } }, audio: false }); video.srcObject = cameraStream; video.style.display = "block"; message.style.display = "none"; await video.play(); startButton.disabled = true; stopButton.disabled = false; status.textContent = "Status: Camera is running."; // Detect unexpected camera disconnection cameraStream.getVideoTracks().forEach(track => { track.addEventListener("ended", () => { status.textContent = "Status: Camera disconnected. Please restart."; startButton.disabled = false; stopButton.disabled = true; video.srcObject = null; video.style.display = "none"; message.style.display = "block"; cameraStream = null; }); }); } catch (error) { console.error("Camera error:", error); if (cameraStream) { cameraStream.getTracks().forEach(track => track.stop()); cameraStream = null; } video.srcObject = null; video.style.display = "none"; message.style.display = "block"; startButton.disabled = false; stopButton.disabled = true; status.textContent = "Camera error: " + error.message; } finally { startingCamera = false; } } function stopCamera() { const video = document.getElementById("camera"); const message = document.getElementById("cameraMessage"); if (cameraStream) { cameraStream.getTracks().forEach(track => track.stop()); cameraStream = null; } video.pause(); video.srcObject = null; video.style.display = "none"; message.style.display = "block"; document.getElementById("startButton").disabled = false; document.getElementById("stopButton").disabled = true; document.getElementById("status").textContent = "Status: Camera stopped."; } // Handle page visibility changes document.addEventListener("visibilitychange", () => { if (document.hidden && cameraStream) { document.getElementById("status").textContent = "Status: Camera paused while page is hidden."; } });
