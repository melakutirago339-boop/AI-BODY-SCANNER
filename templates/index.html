let cameraStream = null; let startingCamera = false; let lastScan = null; async function startCamera() { if (startingCamera) return; const video = document.getElementById("camera"); const message = document.getElementById("cameraMessage"); const status = document.getElementById("status"); startingCamera = true; document.getElementById("startButton").disabled = true; try { cameraStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" }, audio: false }); video.srcObject = cameraStream; video.style.display = "block"; message.style.display = "none"; await video.play(); document.getElementById("stopButton").disabled = false; document.getElementById("scanButton").disabled = false; status.textContent = "Camera is running."; } catch (error) { status.textContent = "Camera Error: " + error.message; document.getElementById("startButton").disabled = false; } finally { startingCamera = false; } } function stopCamera() { if (cameraStream) { cameraStream.getTracks().forEach(track => track.stop()); cameraStream = null; } const video = document.getElementById("camera"); video.pause(); video.srcObject = null; video.style.display = "none"; document.getElementById("cameraMessage").style.display = "block"; document.getElementById("startButton").disabled = false; document.getElementById("stopButton").disabled = true; document.getElementById("scanButton").disabled = true; document.getElementById("status").textContent = "Camera stopped."; } async function scanBody() { const video = document.getElementById("camera"); const status = document.getElementById("status"); if (!cameraStream || video.readyState < 2) { status.textContent = "Start camera first."; return; } status.textContent = "Scanning body..."; try { const canvas = document.createElement("canvas"); canvas.width = video.videoWidth; canvas.height = video.videoHeight; canvas.getContext("2d").drawImage(video, 0, 0); const image = canvas.toDataURL("image/jpeg", 0.85); const response = await fetch("/api/scan", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ image }) }); const data = await response.json(); if (!response.ok || !data.success) { throw new Error(data.error || "Scan failed"); } lastScan = data; document.getElementById("poseStatus").textContent = data.detected ? "Completed" : "Not detected"; document.getElementById("visibility").textContent = data.detected ? "Detected" : "Not detected"; document.getElementById("landmarks").textContent = data.landmarks || 0; document.getElementById("posture").textContent = data.posture || "Not analyzed"; document.getElementById("height").textContent = data.height || "Needs calibration"; document.getElementById("shoulder").textContent = data.shoulder || "Needs calibration"; document.getElementById("proportions").textContent = data.proportions || "Not available"; status.textContent = "Scan completed successfully."; } catch (error) { status.textContent = "Scan Error: " + error.message; console.error(error); } } function downloadReport() { if (!lastScan) { alert("Please scan first."); return; } const blob = new Blob( [JSON.stringify(lastScan, null, 2)], { type: "application/json" } ); const url = URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = "body-scan-report.json"; a.click(); URL.revokeObjectURL(url); }
